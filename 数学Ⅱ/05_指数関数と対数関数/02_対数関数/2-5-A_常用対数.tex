% ============ 1ページ目（表・左） ============
\vspace{-1.6zw}
\heading{〜前時の復習〜}
\vspace{-0.35em}
\begin{reviewbox}{対数関数を含む方程式・不等式}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.02em]
    \item 真数が\fitblankbf{\text{正}}である範囲（\fitblankbf{\text{真数条件}}）を求める.
    \item 対数の定義や, 対数の性質を使って \fitblankbf{式を変形}する.
    \item 対数不等式の底が $a>1$ なら不等号の向きは\fitblankbf{\text{そのまま}},
    $0<a<1$ なら不等号の向きは\fitblankbf{\text{逆}}にする.
    \item 真数条件との \fitblankbf{\text{吟味}}をする.
  \end{enumerate}
\end{reviewbox}

\vspace{0.25em}
大きい数や小さい数は, $10$ の累乗を用いて次のように表せる.
\par\vspace{0.2em}
{\centering
$1620000 = 1.62 \times 10^{6}$,\quad
$0.000618 = 6.18 \times 10^{-4}$
\par}
\vspace{0.15em}
そこで, $10$ を底とする対数を考えて, いろいろな問題に利用してみよう.

\vspace{0.2em}
\heading{常用対数}

\begin{definitionbox}{常用対数}
  $10$ を底とする対数 $\log_{10} M$ を, $M$ の\fitblankbf{\text{常用対数}}という.
\end{definitionbox}

\nextpage

% ============ 2ページ目（表・右） ============
\vspace{-1.4zw}
正の数 $M$ は, 次の形で表すことができる.
\par\vspace{0.2em}
{\centering
$M = a \times 10^{n}$\qquad ただし, $n$ は整数で, $1 \leqq a < 10$
\par}
\vspace{0.2em}
このとき, 対数の性質から
\par\vspace{0.1em}
{\centering
$\log_{10} M = \log_{10} a + \log_{10} 10^{n} = \log_{10} a + n$
\par}

\vspace{0.15em}
\begin{theorembox}{常用対数の表し方}
  正の数 $M$ を $M=a \times 10^{n}$
  （整数 $n$, $1 \leqq a < 10$）と表すとき
  \par\vspace{0.12em}
  {\centering
  $\log_{10} M = \fitblankbf{n} + \fitblankbf{\log_{10} a}$
  \par}
\end{theorembox}

\vspace{0.08em}
\textbf{注意}\quad $1 \leqq a < 10$ であるから, $\log_{10} a$ の値の範囲は
$0 \leqq \log_{10} a < \fitblankbf{1}$ である.

\vspace{0.15em}
巻末の見返しに\fitblankbf{\text{常用対数表}}がある.
表の値は小数第$5$位を四捨五入して小数第$4$位までで, $0.2095$ は $2095$ と書いてある.
\par\vspace{0.1em}
\textbf{見方}\quad 行 $1.6$, 列 $2$ の交点が $2095$ だから
$\log_{10} 1.62 = \fitblankbf{0.2095}$
\par\vspace{0.12em}

\vspace{0.3em}
次の数を $a \times 10^{n}$（$1 \leqq a < 10$）の形で表せ.
\par\vspace{0.15em}
$1620000 = \fitblankbf{1.62 \times 10^{6}}$,\qquad
$0.00162 = \fitblankbf{1.62 \times 10^{-3}}$

\vspace{0.25em}

  \begin{tasks}[label=(\arabic*), label-width=1.6zw, item-indent=2zw, after-item-skip=0.08em](2)
    \task $\log_{10} 1620000$
    \task $\log_{10} 0.00162$
  \end{tasks}
\end{exampleprob}

% ============ 3ページ目（裏・左） ============
\vspace{-1.4zw}
\begin{answerbox}{(1)}[12]
  $1620000 = 1.62 \times 10^{6}$ であるから\\
  $\log_{10} 1620000 = \log_{10} 1.62 + \log_{10} 10^{6}$\\
  $= 0.2095 + 6$
  \boxanswer{$6.2095$}
\end{answerbox}
\vspace{0.12em}
\begin{answerbox}{(2)}[12]
  $0.00162 = 1.62 \times 10^{-3}$ であるから\\
  $\log_{10} 0.00162 = \log_{10} 1.62 + \log_{10} 10^{-3}$\\
  $= 0.2095 - 3$
  \boxanswer{$-2.7905$}
\end{answerbox}

\vspace{0.15em}
\begin{techniquebox}{常用対数の求め方}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.02em]
    \item $M = a \times 10^{n}$ （$1 \leqq a < 10$）の形に変形する.
    \item $\log_{10} M = \fitblankbf{n} + \fitblankbf{\log_{10} a}$ と変形する.
    \item 常用対数表の行と列の交点を読み, 小数第$4$位まで求める.
  \end{enumerate}
\end{techniquebox}

% ============ 4ページ目（裏・右） ============
\vspace{-1.2zw}
\begin{answerbox}{(1)}[7.2]
  $34500 = 3.45 \times 10^{4}$ であるから\\
  $\log_{10} 34500 = \log_{10} 3.45 + 4 = 0.5378 + 4$
  \boxanswer{$4.5378$}
\end{answerbox}
\vspace{0.08em}
\begin{answerbox}{(2)}[7.2]
  $92000 = 9.20 \times 10^{4}$ であるから\\
  $\log_{10} 92000 = \log_{10} 9.20 + 4 = 0.9638 + 4$
  \boxanswer{$4.9638$}
\end{answerbox}
\vspace{0.08em}
\begin{answerbox}{(3)}[7.2]
  $0.00618 = 6.18 \times 10^{-3}$ であるから\\
  $\log_{10} 0.00618 = \log_{10} 6.18 + (-3) = 0.7910 - 3$
  \boxanswer{$-2.2090$}
\end{answerbox}

\vspace{0.35em}
\begin{summarybox}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.08em]
    \item $10$ を底とする対数 $\log_{10} M$ を\fitblankbf{\text{常用対数}}という.
    \item $M = a \times 10^{n}$ （$1 \leqq a < 10$）のとき
          $\log_{10} M = \fitblankbf{n + \log_{10} a}$.
    \item $0 \leqq \log_{10} a < \fitblankbf{1}$ である.
  \end{enumerate}
\end{summarybox}
