\heading{〜前時の復習〜}
\vspace{-0.55em}
\begin{reviewbox}{対数関数とそのグラフ}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.02em]
    \item $y=\log_{a} x$ のグラフは $y=a^{x}$ と直線 $y=\fitblankbf{x}$ に関して対称.
    \item 点 $(1,\ \fitblankbf{0})$, $(a,\ \fitblankbf{1})$ を通り, \fitblankbf{y \text{軸}}を漸近線とする.
    \item $a>1$ のとき\fitblankbf{\text{増加関数}}, $0<a<1$ のとき\fitblankbf{\text{減少関数}}.
  \end{enumerate}
\end{reviewbox}

\heading{対数関数を含む方程式}
\vspace{-0.25em}
対数方程式も指数方程式と同様に, 今まで学習してきた知識を使って \fitblankbf{x = \text{○}} の形に変形する.

\begin{exampleprob}{6}
  次の方程式を解け。\\
  (1)\quad $\log_{2} x = 3$
\end{exampleprob}
\begin{answerbox}{(1)}[3.8]
  対数の定義から, \qquad $x=2^{3}=8$
\end{answerbox}

\vspace{1.2em}
\begin{techniquebox}{対数方程式の解き方}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.02em]
    \item 真数が\fitblankbf{\text{正}}である範囲を求める.
    \item 対数の定義 $(\log_{a} x = p \Longleftrightarrow x = \fitblankbf{a^{p}})$ や,

    対数の性質 $(\log_{a} M+\log_{a} N=\fitblankbf{\log_{a} MN}$ など) を使って式を変形する.
    \item 求めた解が, \fitblankbf{真数条件}をみたすか確かめる.
  \end{enumerate}
\end{techniquebox}

\noindent{\bfseries 応用例題\,1}\quad
方程式 $\log_{3} x+\log_{3}(x-8)=2$ を解け。
\vspace{0.06em}

\begin{answerbox}{}[13]
  \makebox[15zw][l]{真数が正であるから,}$x>0$ かつ $x>8$ \\
  \makebox[15zw][l]{すなわち,}$x>8$ \quad ……①\\
  \makebox[15zw][l]{方程式を変形すると,}$\begin{aligned}[t]
    \log_{3} x(x-8) &= 2 \\
    x(x-8) &= 3^{2} \\
    (x+1)(x-9) &= 0
  \end{aligned}$
  \makebox[15zw][l]{①を満たすのは} $x=9$
\end{answerbox}

\begin{practiceprob}{22}

  \begin{tasks}[label=(\arabic*), label-width=1.6zw, item-indent=2zw, after-item-skip=0.06em](2)

    \log_{4} x(x-6) &= 2 \\
    x(x-6) &= 4^{2} \\
  \end{aligned}$
  \makebox[15zw][l]{(*)を満たすのは} $x=8$
  \end{answerbox}

% ============ 3ページ目（裏・左） ============
\heading{対数関数を含む不等式}
\vspace{-0.25em}
\begin{exampleprob}{6}
  (2)\quad $\log_{2} x \leqq 3$
\end{exampleprob}
\begin{answerbox}{(2)}[9]
  \makebox[15zw][l]{真数が正であるから,}$x>0$ \quad ……①\\
  \makebox[15zw][l]{不等式を変形すると,}$\log_{2} x \leqq \log_{2} 2^{3}$ \\
  \makebox[15zw][l]{底 $2$ は $1$ より大きいから,}$x \leqq 8$ \quad ……② \\
  \makebox[15zw][l]{①, ②の共通範囲を求めて,} $0<x\leqq 8$
\end{answerbox}
\vspace{-0.5em}

\begin{techniquebox}{対数不等式の解き方}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.02em]
    \item \fitblankbf{\text{真数条件}}を求める.
    \item 底の変換公式などを使って, 両辺を同じ底の対数にする.
    \item 底 $a>1$ なら不等号の向きは\fitblankbf{\text{そのまま}}, $0<a<1$ なら不等号の向きは\fitblankbf{\text{逆}}にする.
    \item 真数条件との\fitblankbf{\text{共通範囲}}を求める.
  \end{enumerate}
\end{techniquebox}
\vspace{-0.5em}

\noindent{\bfseries 応用例題\,2}\quad
不等式 $\log_{\frac{1}{3}}(x-1)>2$ を解け。
\vspace{0.06em}

\begin{answerbox}{}[10]
  \makebox[15zw][l]{真数が正であるから,}$x>1$ \quad ……①\\
  \makebox[15zw][l]{不等式を変形すると,}$\log_{\frac{1}{3}}(x-1) > \log_{\frac{1}{3}}\dfrac{1}{9}$ \\
  \makebox[15zw][l]{底 $\frac{1}{3}$ は $1$ より小さいから,}$x-1 < \dfrac{1}{9}$ \quad ……②\\
  \makebox[15zw][l]{①, ②を満たすのは} $1<x<\dfrac{10}{9}$
\end{answerbox}

\begin{practiceprob}{23}
  次の不等式を解け。
  \begin{tasks}[label=(\arabic*), label-width=1.6zw, item-indent=2zw, after-item-skip=0.04em](2)
    \task $\log_{2}(x-3)<4$
    \task $\log_{\frac{1}{2}}(x+2) \geqq 3$
  \end{tasks}
\end{practiceprob}
\begin{answerbox}{(1)}[10]
  \makebox[15zw][l]{真数が正であるから,}$x>3$ \quad ……①\\
  \makebox[15zw][l]{不等式を変形すると,}$\log_{2}(x-3) < \log_{2} 2^{4}$ \\
  \makebox[15zw][l]{底 $2$ は $1$ より大きいから,}$x<19$ \quad ……②\\
  \makebox[15zw][l]{①, ②の共通範囲を求めて,} $3<x<19$
\end{answerbox}
\vspace{0.08em}
\begin{answerbox}{(2)}[10]
  \makebox[15zw][l]{真数が正であるから,}$x>-2$ \quad ……①\\
  \makebox[15zw][l]{不等式を変形すると,}$\log_{\frac{1}{2}}(x+2) \geqq \log_{\frac{1}{2}}\left(\dfrac{1}{2}\right)^{3}$ \\
  \makebox[15zw][l]{底 $\frac{1}{2}$ は $1$ より小さいから,}$x\leqq -\dfrac{15}{8}$ \quad ……②\\
  \makebox[15zw][l]{①, ②の共通範囲を求めて,} $-2<x\leqq -\dfrac{15}{8}$
\end{answerbox}

\vspace{0.3em}
\begin{summarybox}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.06em]
    \item 真数が\fitblankbf{\text{正}}である範囲（\fitblankbf{\text{真数条件}}）を求める.
    \item 対数の定義や, 対数の性質を使って \fitblankbf{式を変形}する.
    \item 対数不等式の底が $a>1$ なら不等号の向きは\fitblankbf{\text{そのまま}},

    $0<a<1$ なら不等号の向きは\fitblankbf{\text{逆}}にする.
    \item 真数条件との \fitblankbf{\text{吟味}}をする.
  \end{enumerate}
\end{summarybox}

\noindent{\bfseries 演習問題}\quad
次の方程式を解け。
\par\vspace{-0.22em}
\begin{tasks}[label=(\arabic*), label-width=1.6zw, item-indent=2zw, after-item-skip=0.04em](2)
  \task $\log_{6} x+\log_{6}(x+1)=1$
  \task $2\log_{2}(x-3)=\log_{2}(5-x)$
\end{tasks}
\vspace{0.04em}
\begin{answerbox}{(1)}[24.5]
  \makebox[15zw][l]{真数が正であるから,}$x>0$ かつ $x>-1$ \\
  \makebox[15zw][l]{すなわち,}$x>0$ \quad ……①\\
  \makebox[15zw][l]{方程式を変形すると,}$\begin{aligned}[t]
    \log_{6} x(x+1) &= 1 \\
    x(x+1) &= 6 \\
    (x+3)(x-2) &= 0
  \end{aligned}$
  \makebox[15zw][l]{①を満たすのは} $x=2$
\end{answerbox}
\vspace{0.12em}
\begin{answerbox}{(2)}[24.5]
  \makebox[15zw][l]{真数が正であるから,}$x>3$ かつ $x<5$ \\
  \makebox[15zw][l]{すなわち,}$3<x<5$ \quad ……①\\
  \makebox[15zw][l]{方程式を変形すると,}$\begin{aligned}[t]
    \log_{2}(x-3)^{2} &= \log_{2}(5-x) \\
    (x-3)^{2} &= 5-x \\
    (x-1)(x-4) &= 0
  \end{aligned}$
  \makebox[15zw][l]{①を満たすのは} $x=4$
\end{answerbox}

\noindent{\bfseries 演習問題}\quad
次の不等式を解け。
\par\vspace{-0.22em}
\begin{tasks}[label=(\arabic*), label-width=1.6zw, item-indent=2zw, after-item-skip=0.04em](2)
  \task $\log_{2}(x+1)>\log_{2}(2x-5)$
  \task $2\log_{\frac{1}{3}}(x-1)<\log_{\frac{1}{3}}(13-x)$
\end{tasks}
\vspace{0.04em}
\begin{answerbox}{(1)}[24.5]
  \makebox[15zw][l]{真数が正であるから,}$x>-1$ かつ $x>\dfrac{5}{2}$ \\
  \makebox[15zw][l]{すなわち,}$x>\dfrac{5}{2}$ \quad ……①\\
  \makebox[15zw][l]{底 $2$ は $1$ より大きいから,}$x+1>2x-5$ \\
  \makebox[15zw][l]{} $x<6$ \quad ……②\\
  \makebox[15zw][l]{①, ②を満たすのは} $\dfrac{5}{2}<x<6$
\end{answerbox}
\vspace{0.12em}
\begin{answerbox}{(2)}[24.5]
  \makebox[15zw][l]{真数が正であるから,}$x>1$ かつ $x<13$ \\
  \makebox[15zw][l]{すなわち,}$1<x<13$ \quad ……①\\
  \makebox[15zw][l]{不等式を変形すると,}$\begin{aligned}[t]
    \log_{\frac{1}{3}}(x-1)^{2} &< \log_{\frac{1}{3}}(13-x) \\
    (x-1)^{2} &> 13-x \\
    (x-4)(x+3) &> 0 \quad \text{……②}
  \end{aligned}$
  \makebox[15zw][l]{①, ②を満たすのは} $4<x<13$
\end{answerbox}
