\vspace{0.2em}
\heading{常用対数の応用}
自然数 $N$ の桁数と, 常用対数 $\log_{10} N$ の関係を調べよう.

\vspace{0.15em}
\noindent{\bfseries 例\,13}\quad
自然数 $N$ が $3$ 桁の数のとき, $N$ のとり得る値の範囲を考える.
\par\vspace{0.12em}
$3$ 桁の自然数のうち, 最小の数は $100$, 最大の数は $999$ であるから
\par\vspace{0.1em}
{\centering
$100 \leqq N < 1000$\\[0.45em]
すなわち\qquad $10^{\fitblankbf{2}} \leqq N < 10^{\fitblankbf{3}}$
\par}
\vspace{0.12em}
両辺の常用対数をとると
\par\vspace{0.08em}
{\centering
$\log_{10} 10^{2} \leqq \log_{10} N < \log_{10} 10^{3}$\\[0.45em]
よって\qquad $\fitblankbf{2} \leqq \log_{10} N < \fitblankbf{3}$
\par}

\vspace{5em}
\begin{theorembox}{桁数と常用対数}
  一般に、自然数 $N$ が $k$ について次が成り立つ.
  \par\vspace{0.12em}
  {\centering
  $N$ が \fitblankbf{k} 桁の数\qquad\
  $\Longleftrightarrow$\qquad
  $\fitblankbf{k-1} \leqq \log_{10} N < \fitblankbf{k}$
  \par}
\end{theorembox}

\begin{exampleprob}{7}
  $3^{20}$ は何桁の数か。ただし, $\log_{10} 3 = 0.4771$ とする。
\end{exampleprob}

\begin{answerbox}{}[13]
  $\log_{10} 3^{20} = 20\log_{10} 3 = 20 \times 0.4771 = 9.5420$\\
  $9 < 9.5420 < 10$ であるから\\
  $9\log_{10} 10 \leqq \log_{10} 3^{20} < 10\log_{10} 10$\\
  $\log_{10} 10^{9} \leqq \log_{10} 3^{20} < \log_{10} 10^{10}$\\
  $10^{9} < 3^{20} < 10^{10}$
  \boxanswer{$10$ 桁}
\end{answerbox}

\vspace{1em}
\noindent{\bfseries 応用例題\,3}\quad
$2^{n}$ が $10$ 桁の数となるような自然数 $n$ をすべて求めよ。
ただし, $\log_{10} 2 = 0.3010$ とする。

\vspace{1em}
\begin{answerbox}{}[18]
  \makebox[17zw][l]{$2^{n}$ が $10$ 桁の数のとき}$10^{9} \leqq 2^{n} < 10^{10}$\\
  \makebox[17zw][l]{常用対数をとると}$9 \leqq n\log_{10} 2 < 10$\\
  \makebox[17zw][l]{$\log_{10} 2 = 0.3010 > 0$ であるから}$\dfrac{9}{\log_{10} 2} \leqq n < \dfrac{10}{\log_{10} 2}$ \quad ……①\\
  \vspace{0.8em}
  \makebox[17zw][l]{ここで}$\frac{9}{\log_{10} 2}=29.9\cdots$,\\
  \vspace{0.8em}
  \makebox[17zw][l]{}$\frac{10}{\log_{10} 2}=33.2\cdots$\\
  \vspace{0.8em}
  \makebox[17zw][l]{よって、①を満たす自然数 $n$ は}$n=30,\ 31,\ 32,\ 33$
\end{answerbox}

\begin{practiceprob}{26}
  $3^{n}$ が $8$ 桁の数となるような自然数 $n$ をすべて求めよ。
  ただし, $\log_{10} 3 = 0.4771$ とする。
\end{practiceprob}

\begin{answerbox}{}[16]
  \makebox[17zw][l]{$3^{n}$ が $8$ 桁の数のとき}$10^{7} \leqq 3^{n} < 10^{8}$\\
  \makebox[17zw][l]{常用対数をとると}$7 \leqq n\log_{10} 3 < 8$\\
  \makebox[17zw][l]{$\log_{10} 3 = 0.4771 > 0$ であるから}$\dfrac{7}{\log_{10} 3} \leqq n < \dfrac{8}{\log_{10} 3}$ \quad ……①\\
  \vspace{0.45em}
  \makebox[17zw][l]{ここで}$\frac{7}{\log_{10} 3}=14.6\cdots$,\\
  \vspace{0.45em}
  \makebox[17zw][l]{}$\frac{8}{\log_{10} 3}=16.7\cdots$\\
  \vspace{0.45em}
  \makebox[17zw][l]{よって、①を満たす自然数 $n$ は}$n=15,\ 16$
\end{answerbox}

\vspace{0.2em}
$0 < M < 1$ である小数 $M$ と常用対数 $\log_{10} M$ の関係を調べよう.

\vspace{0.08em}
たとえば, $M$ の小数第 $3$ 位に初めて $0$ でない数字が現れるとは,
$M$ が次を満たすということである.
\par\vspace{0.06em}
{\centering
$0.001 \leqq M < 0.01$\\[0.3em]
すなわち\qquad $10^{\fitblankbf{-3}} \leqq M < 10^{\fitblankbf{-2}}$
\par}
\vspace{0.06em}
$0.001 = 10^{-3}$,\ $0.01 = 10^{-2}$ である.
両辺の常用対数をとると
\par\vspace{0.04em}
{\centering
$\fitblankbf{-3} \leqq \log_{10} M < \fitblankbf{-2}$
\par}

\vspace{0.2em}
\begin{theorembox}{小数の位と常用対数}
  一般に、$0 < M < 1$ である小数 $M$ と自然数 $k$ について次が成り立つ.
  \par\vspace{0.1em}
  {\centering
  小数第 \fitblankbf{k} 位に初めて $0$ でない数字が現れる\\[0.3em]
  $\Longleftrightarrow$\qquad
  $\fitblankbf{-k} \leqq \log_{10} M < \fitblankbf{-(k-1)}$
  \par}
\end{theorembox}

\vspace{-1.2zw}
\begin{exampleprob}{8}
  $\left(\dfrac{1}{3}\right)^{30}$ を小数で表したとき, 小数第何位に初めて $0$ でない数字が
  現れるか。ただし, $\log_{10} 3 = 0.4771$ とする。
\end{exampleprob}

\begin{answerbox}{}[11]
  $\log_{10}\!\left(\dfrac{1}{3}\right)^{30}
    = 30\log_{10}\dfrac{1}{3}
    = -30\log_{10} 3
    = -30 \times 0.4771 = -14.313$\\
  $-15 < -14.313 < -14$ であるから\\
  $10^{-15} < \left(\dfrac{1}{3}\right)^{30} < 10^{-14}$
  \boxanswer{小数第 $15$ 位}
\end{answerbox}

\vspace{0.1em}
\begin{practiceprob}{27}
  $\left(\dfrac{1}{2}\right)^{20}$ を小数で表したとき, 小数第何位に初めて $0$ でない数字が
  現れるか。ただし, $\log_{10} 2 = 0.3010$ とする。
\end{practiceprob}

\begin{answerbox}{}[11]
  $\log_{10}\!\left(\dfrac{1}{2}\right)^{20}
    = 20\log_{10}\dfrac{1}{2}
    = -20\log_{10} 2
    = -20 \times 0.3010 = -6.0200$\\
  $-7 < -6.0200 < -6$ であるから\\
  $10^{-7} < \left(\dfrac{1}{2}\right)^{20} < 10^{-6}$
  \boxanswer{小数第 $7$ 位}
\end{answerbox}

\vspace{0.15em}
\begin{summarybox}
  \begin{enumerate}[label=\textbf{\arabic*}, leftmargin=1.5zw, itemsep=0.08em]
    \item $N$ が \fitblankbf{k} 桁の数 $\Longleftrightarrow$
      $\fitblankbf{k-1} \leqq \log_{10} N < \fitblankbf{k}$.
    \item 小数第 \fitblankbf{k} 位に初めて $0$ でない数字が現れる $\Longleftrightarrow$\\
      $\fitblankbf{-k} \leqq \log_{10} M < \fitblankbf{-(k-1)}$.
  \end{enumerate}
\end{summarybox}
